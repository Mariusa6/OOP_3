19double studentas::mediana(const Vector<int>& nd) {
20 if (nd.empty())
21 return 0.0;
22 Vector<int> sortedNd = nd;
23 std::sort(sortedNd.begin(), sortedNd.end());
24 size_t size = sortedNd.size();
25 if (size % 2 == 0)
26 return (sortedNd[size / 2 - 1] + sortedNd[size / 2]) / 2.0;
27 else
28 return sortedNd[size / 2];
29}

296// -------------------------------------------------------
297// parseFromLine — GREITAS parsinimas failo skaitymui
298// Nenaudoja std::istringstream, naudoja std::from_chars
299// -------------------------------------------------------
300void studentas::parseFromLine(const std::string& line, int ndCount, int lineNumber)
301{
302 const char* p = line.data();
303 const char* end = p + line.size();
304
305 auto skipSpace = [&]() {
306 while (p < end && (*p == ' ' || *p == '\t' || *p == '\r')) ++p;
307 };
308
309 // --- Vardas ---
310 skipSpace();
311 const char* start = p;
312 while (p < end && *p != ' ' && *p != '\t' && *p != '\r') ++p;
313 if (start == p)
314 throw std::runtime_error(vietosTekstas(lineNumber) + " trūksta vardo.");
315 vardas_.assign(start, p - start);
316
317 // --- Pavardė ---
318 skipSpace();
319 start = p;
320 while (p < end && *p != ' ' && *p != '\t' && *p != '\r') ++p;
321 if (start == p)
322 throw std::runtime_error(vietosTekstas(lineNumber) + " trūksta pavardės.");
323 pavarde_.assign(start, p - start);
324
325 // --- Pažymiai ---
326 auto readInt = [&](const char* ko) -> int {
327 skipSpace();
328 int v = 0;
329 auto res = std::from_chars(p, end, v);
330 if (res.ec != std::errc())
331 throw std::runtime_error(vietosTekstas(lineNumber)
332 + " trūksta " + ko + " pažymio.");
333 p = res.ptr;
334 return v;
335 };
336
337 nd_.clear();
338 nd_.reserve(ndCount);
339
340 for (int i = 0; i < ndCount; ++i) {
341 int v = readInt("namų darbo");
342 if (!pazymysTinkamas(v))
343 throw std::runtime_error(ribuKlaida(lineNumber, "namų darbo", v));
344 nd_.push_back(v);
345 }
346
347 egzaminas_ = readInt("egzamino");
348 if (!pazymysTinkamas(egzaminas_))
349 throw std::runtime_error(ribuKlaida(lineNumber, "egzamino", egzaminas_));
350}

352// -------------------------------------------------------
353// readStudentas — įvestis per std::istream
354// -------------------------------------------------------
355std::istream& studentas::readStudentas(std::istream& is, int ndCount, int lineNumber) {
356 if (!(is >> vardas_ >> pavarde_))
357 throw std::runtime_error(vietosTekstas(lineNumber)
358 + " trūksta vardo arba pavardės.");
359
360 nd_.clear();
361
362 if (ndCount > 0) {
363 nd_.reserve(ndCount);
364 for (int i = 0; i < ndCount; ++i) {
365 int p;
366 if (!(is >> p))
367 throw std::runtime_error(vietosTekstas(lineNumber)
368 + " trūksta namų darbo pažymio.");
369 if (!pazymysTinkamas(p))
370 throw std::runtime_error(ribuKlaida(lineNumber, "namų darbo", p));
371 nd_.push_back(p);
372 }
373 if (!(is >> egzaminas_))
374 throw std::runtime_error(vietosTekstas(lineNumber)
375 + " trūksta egzamino pažymio.");
376 }
377 else {
378 // Nežinomas skaičius — skaitome iki galo, paskutinis yra egzaminas
379 int p;
380 while (is >> p) {
381 if (!pazymysTinkamas(p))
382 throw std::runtime_error(ribuKlaida(lineNumber, "namų darbo", p));
383 nd_.push_back(p);
384 }
385 if (nd_.empty())
386 throw std::runtime_error(vietosTekstas(lineNumber)
387 + " trūksta egzamino pažymio.");
388
389 egzaminas_ = nd_.back();
390 nd_.pop_back();
391 is.clear(); // eilutės pabaiga nėra klaida šiame režime
392 }
393
394 if (!pazymysTinkamas(egzaminas_))
395 throw std::runtime_error(ribuKlaida(lineNumber, "egzamino", egzaminas_));
396
397 return is;
398}

406// -------------------------------------------------------
407void studentas::appendListTo(std::string& out) const
408{
409 out.append(vardas_);
410 if (vardas_.size() < 25) out.append(25 - vardas_.size(), ' ');
411
412 out.append(pavarde_);
413 if (pavarde_.size() < 25) out.append(25 - pavarde_.size(), ' ');
414
415 char buf[16];
416 auto appendInt = [&](int v) {
417 auto res = std::to_chars(buf, buf + sizeof(buf), v);
418 size_t len = static_cast<size_t>(res.ptr - buf);
419 if (len < 10) out.append(10 - len, ' ');
420 out.append(buf, len);
421 };
422
423 for (int nd : nd_)
424 appendInt(nd);
425 appendInt(egzaminas_);
426
427 out.push_back('\n');
428}
